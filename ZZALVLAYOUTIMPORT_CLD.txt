  PUBLIC SECTION.

    METHODS:

        FOR EVENT node_double_click
        OF cl_gui_alv_tree_simple
        IMPORTING
          index_outtab
          grouplevel.

    METHODS:
      item_double_click
        FOR EVENT item_double_click
        OF cl_gui_alv_tree_simple

          index_outtab
          grouplevel.

ENDCLASS.
